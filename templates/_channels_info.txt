<?php
header('Content-Type: application/json');

function getBaseUrl() {
    if (php_sapi_name() === 'cli') {
        return 'http://localhost';
    }
    $forwardedProto = $_SERVER['HTTP_X_FORWARDED_PROTO'] ?? '';
    $isHttps = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off')
        || strcasecmp($forwardedProto, 'https') === 0;
    $protocol = $isHttps ? "https" : "http";
    $host = $_SERVER['HTTP_HOST'];
    $scriptDir = dirname($_SERVER['SCRIPT_NAME']);
    $projectRoot = preg_replace('#/channels$#', '', $scriptDir);
    if ($projectRoot === $scriptDir) {
        $projectRoot = $scriptDir;
    }
    return $protocol . "://" . $host . $projectRoot;
}

$baseDir = __DIR__;
$channels = [];
$baseUrl = getBaseUrl();
$channelFilter = $_GET['channel'] ?? null;

// Scan all subdirectories in channels folder (or just the requested one)
$items = $channelFilter !== null ? [$channelFilter] : scandir($baseDir);
foreach ($items as $item) {
    if ($item === '.' || $item === 'index.php' || $item === 'info.php') continue;
    $channelPath = $baseDir . DIRECTORY_SEPARATOR . $item;
    if (is_dir($channelPath)) {
        $specsPath = $channelPath . DIRECTORY_SEPARATOR . 'specs';
        if (is_dir($specsPath)) {
            $specFiles = glob($specsPath . DIRECTORY_SEPARATOR . '*.json');
            $specs = [];
            foreach ($specFiles as $specFile) {
                $jsonContent = file_get_contents($specFile);
                $data = json_decode($jsonContent, true);
                if ($data) {
                    $fileName = basename($specFile);
                    $pathFromSpec = $data['path'] ?? $fileName;
                    $endpointName = preg_replace('/\.json$/', '', $pathFromSpec);
                    $endpointPath = $baseUrl . '/channels/' . $item . '/' . $endpointName;

                    $form_endpoint = "";
                    if($data["method"] == "POST") {
                        $form_endpoint = $endpointPath . '/form';
                    }

                    $specs[] = [
                        'channel' => $item,
                        'endpoint_name' => $endpointName,
                        'method' => $data['method'] ?? null,
                        'endpoint_path' => $endpointPath,
                        'api_spec_endpoint' => $endpointPath . '/api-spec',
                        'form_endpoint' => $form_endpoint,
                        'description' => $data['description'] ?? null,
                    ];
                }
            }
            if (!empty($specs)) {
                $channels[$item] = $specs;
            }
        }
    }
}

// Build user_stories dynamically from each channel's 'user_stories' in config.json
// (config.json is regenerated from the Superlindey project spec, so this stays in sync
// with whatever is set on the channel via update_channel).
$user_stories = [];
$configPath = $baseDir . '/../Library/config.json';
if (file_exists($configPath)) {
    $projectConfig = json_decode(file_get_contents($configPath), true);
    foreach ($projectConfig['channels'] ?? [] as $channelConfig) {
        $channelName = $channelConfig['channel_name'] ?? null;
        if ($channelName !== null && ($channelFilter === null || $channelName === $channelFilter)) {
            $user_stories[$channelName] = $channelConfig['user_stories'] ?? '';
        }
    }
}

        "step_1" => "Read 'user_stories' (sibling of 'how_to_use') first to understand each channel's purpose and intended user flows — this should drive FE structure/UX decisions, not just the endpoint list.",

$output = array_merge($usage_guide, $channels);
echo json_encode($output, JSON_PRETTY_PRINT);
